using System.Collections;
using System.IO;
using MetroSort.Game;
using MetroSort.Persistence;
using MetroSort.Puzzle;
using NUnit.Framework;
using UnityEngine;
using UnityEngine.TestTools;

namespace MetroSort.Tests
{
    // Development helper: copy into Assets/MetroSort/Tests/PlayMode as .cs (and add the imageconversion module)
    // to render 1080x1920 frames of the running game into Logs/shots.
    public class ZzScreenshotTemp
    {
        private static IEnumerator Shot(string name)
        {
            yield return null;
            Camera cam = GameApp.Instance.Station.Cam;
            int w = 1080, h = 1920;
            foreach (Canvas c in Object.FindObjectsByType<Canvas>())
            {
                c.renderMode = RenderMode.ScreenSpaceCamera;
                c.worldCamera = cam;
                c.planeDistance = 1f;
            }
            var rt = new RenderTexture(w, h, 24);
            cam.targetTexture = rt;
            yield return null;
            yield return null;
            cam.Render();
            RenderTexture.active = rt;
            var tex = new Texture2D(w, h, TextureFormat.RGB24, false);
            tex.ReadPixels(new Rect(0, 0, w, h), 0, 0);
            tex.Apply();
            RenderTexture.active = null;
            cam.targetTexture = null;
            Directory.CreateDirectory("Logs/shots");
            File.WriteAllBytes("Logs/shots/" + name + ".png", tex.EncodeToPNG());
            Debug.Log("SHOT " + name);
        }

        [UnityTest]
        public IEnumerator Capture()
        {
            GameApp app = GameApp.Create(new InMemoryFileStore());
            app.Services.Save.Data.HighestUnlocked = 9999;
            yield return null; yield return null;
            yield return Shot("warmup");
            foreach (int id in new[] { 101, 25, 45, 65, 85 })
            {
                app.Game.StartLevel(id);
                yield return new WaitForSeconds(1.5f);
                yield return Shot("level" + id);
            }
            // Reveal check: play three moves on a mystery level.
            app.Game.StartLevel(102);
            yield return new WaitForSeconds(1.2f);
            for (int i = 0; i < 3; i++)
            {
                Move m; PuzzleSolver.TryGetHint(app.Game.Board.State, out m);
                while (app.Game.TryMove(m) == MoveError.PlatformBusy) yield return null;
                yield return new WaitForSeconds(0.5f);
            }
            yield return Shot("level102_after_moves");
            app.GoToMenu();
            yield return null;
            app.UI.OpenLevels();
            yield return new WaitForSeconds(0.4f);
            yield return Shot("levels_page");
            Object.Destroy(app.gameObject);
            yield return null;
        }
    }
}
